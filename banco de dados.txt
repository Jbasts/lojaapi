-- =====================================================
-- 1. USUÁRIO ADMINISTRADOR
-- =====================================================

CREATE TABLE usuarios_admin (
    id BIGSERIAL PRIMARY KEY,

    nome VARCHAR(100) NOT NULL,
    sobrenome VARCHAR(100) NOT NULL,

    email VARCHAR(255) NOT NULL UNIQUE,

    senha_hash TEXT NOT NULL,

    ativo BOOLEAN NOT NULL DEFAULT TRUE,

    criado_em TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
);


-- =====================================================
-- 2. CLIENTES
-- Tela Cadastro Usuários
-- =====================================================

CREATE TABLE clientes (
    id BIGSERIAL PRIMARY KEY,

    nome VARCHAR(100) NOT NULL,
    sobrenome VARCHAR(100) NOT NULL,

    telefone VARCHAR(20),

    email VARCHAR(255),

    cpf VARCHAR(11),

    cep VARCHAR(8),

    bairro VARCHAR(120),

    rua VARCHAR(200),

    numero_endereco VARCHAR(20),

    complemento VARCHAR(120),

    ativo BOOLEAN NOT NULL DEFAULT TRUE,

    criado_em TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
);


CREATE UNIQUE INDEX uq_clientes_cpf
ON clientes(cpf)
WHERE cpf IS NOT NULL;


CREATE UNIQUE INDEX uq_clientes_email
ON clientes(email)
WHERE email IS NOT NULL;


-- =====================================================
-- 3. PRODUTOS DE ESTOQUE
-- =====================================================

CREATE TABLE produtos_estoque (
    id BIGSERIAL PRIMARY KEY,

    nome VARCHAR(200) NOT NULL,

    codigo_barras VARCHAR(50) NOT NULL UNIQUE,

    unidade VARCHAR(20) NOT NULL DEFAULT 'UN',

    controla_validade BOOLEAN NOT NULL DEFAULT TRUE,

    estoque_minimo NUMERIC(14,3) NOT NULL DEFAULT 0,

    ativo BOOLEAN NOT NULL DEFAULT TRUE,

    criado_em TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
);


-- =====================================================
-- 4. PRODUTOS DE VENDA
--
-- PRODUCAO = Trufa
-- REVENDA = Salgadinho
-- =====================================================

CREATE TABLE produtos_venda (
    id BIGSERIAL PRIMARY KEY,

    produto_estoque_id BIGINT,

    nome VARCHAR(150) NOT NULL,

    sabor VARCHAR(120),

    tipo VARCHAR(20) NOT NULL,

    unidade VARCHAR(20) NOT NULL DEFAULT 'UN',

    preco_venda NUMERIC(14,2) NOT NULL,

    ativo BOOLEAN NOT NULL DEFAULT TRUE,

    criado_em TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_produto_venda_estoque
        FOREIGN KEY (produto_estoque_id)
        REFERENCES produtos_estoque(id),

    CONSTRAINT ck_produto_venda_tipo
        CHECK (tipo IN ('PRODUCAO', 'REVENDA')),

    CONSTRAINT ck_produto_venda_preco
        CHECK (preco_venda >= 0)
);


-- =====================================================
-- 5. FICHA TÉCNICA

CREATE TABLE fichas_tecnicas (

    nome VARCHAR(150) NOT NULL,

-- =====================================================
-- 7. PEDIDOS
-- =====================================================

CREATE TABLE pedidos (
    id BIGSERIAL PRIMARY KEY,

    numero VARCHAR(12) NOT NULL UNIQUE,

    cliente_id BIGINT,

    nome_cliente_avulso VARCHAR(200),

    status VARCHAR(30) NOT NULL DEFAULT 'ABERTO',

    data_pedido TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,

    data_conclusao TIMESTAMP,

    observacao TEXT,

Views

Essas são importantes porque vão facilitar bastante o React.

CREATE VIEW vw_pedidos_totais AS

SELECT
    p.id AS pedido_id,

    p.numero,

    COALESCE(
        CONCAT(c.nome, ' ', c.sobrenome),
        p.nome_cliente_avulso
    ) AS cliente,

    p.status,

    p.data_pedido,

    p.data_conclusao,

    COALESCE(
        SUM(ip.quantidade * ip.valor_unitario),
        0
    ) AS valor_bruto,

    COALESCE(
        SUM(
            CASE

                WHEN ip.tipo_desconto = 'VALOR'
                THEN ip.desconto

                WHEN ip.tipo_desconto = 'PERCENTUAL'
                THEN
                    (
                        ip.quantidade
                        * ip.valor_unitario
                    )
                    *
                    (
                        ip.desconto / 100
                    )

                ELSE 0

            END
        ),
        0
    ) AS desconto_total,

    COALESCE(
        SUM(

            (ip.quantidade * ip.valor_unitario)

            -

            CASE

                WHEN ip.tipo_desconto = 'VALOR'
                THEN ip.desconto

                WHEN ip.tipo_desconto = 'PERCENTUAL'
                THEN
                    (
                        ip.quantidade
                        * ip.valor_unitario
                    )
                    *
                    (
                        ip.desconto / 100
                    )

                ELSE 0

            END
        ),
        0
    ) AS valor_total

FROM pedidos p

LEFT JOIN clientes c
ON c.id = p.cliente_id

LEFT JOIN itens_pedido ip
ON ip.pedido_id = p.id

GROUP BY
    p.id,
    p.numero,
    c.nome,
    c.sobrenome,
    p.nome_cliente_avulso,
    p.status,
    p.data_pedido,
    p.data_conclusao;

Essa será muito útil para sua Tela Estoque.


CREATE VIEW vw_estoque AS

SELECT

    le.id AS lote_id,

    pe.id AS produto_id,

    pe.nome,

    pe.codigo_barras,

    pe.unidade,

    le.quantidade_atual,

    le.custo_unitario,

    (
        le.quantidade_atual
        *
        le.custo_unitario
    ) AS valor_total,

    le.validade,

    CASE

        WHEN le.validade IS NULL
        THEN 'SEM VALIDADE'

        WHEN le.validade < CURRENT_DATE
        THEN 'INVALIDO'

        ELSE 'VALIDO'

    END AS status

FROM lotes_estoque le

INNER JOIN produtos_estoque pe
ON pe.id = le.produto_estoque_id;

CREATE VIEW vw_vendas AS

SELECT
    p.id AS pedido_id,
    p.numero AS numero_pedido,

    c.id AS cliente_id,

    CONCAT(
        c.nome,
        ' ',
        c.sobrenome
    ) AS cliente,

    pv.id AS produto_id,
    pv.nome AS produto,
    pv.sabor,

    ip.quantidade,
    ip.valor_unitario,

    ip.desconto_tipo,
    ip.desconto_valor,

    (
        ip.quantidade
        *
        ip.valor_unitario
    ) AS valor_bruto,

    CASE

        WHEN ip.desconto_tipo = 'PERCENTUAL'
        THEN
            (
                ip.quantidade
                *
                ip.valor_unitario
            )
            *
            (
                ip.desconto_valor / 100
            )

        WHEN ip.desconto_tipo = 'VALOR'
        THEN ip.desconto_valor

        ELSE 0

    END AS valor_desconto,

    (
        ip.quantidade
        *
        ip.valor_unitario
    )
    -
    CASE

        WHEN ip.desconto_tipo = 'PERCENTUAL'
        THEN
            (
                ip.quantidade
                *
                ip.valor_unitario
            )
            *
            (
                ip.desconto_valor / 100
            )

        WHEN ip.desconto_tipo = 'VALOR'
        THEN ip.desconto_valor

        ELSE 0

    END AS valor_final,

    p.data_pedido,
    p.concluido_em

FROM pedidos p

INNER JOIN clientes c
